\chapter{Finite trees and the total progeny}
\label{ch:progeny}

\begin{lemma}[The point mass of a finite tree]
\label{lem:point-mass}
\uses{def:sampleMeasure}
\lean{BranchingProcess.sampleMeasure_agreeOn,
BranchingProcess.ofReal_pow_le_sampleMeasure_sample_eq}
\leanok
Two fields agreeing at the vertices of one of their samples cut out the same tree. Hence the
probability that the sample is a prescribed finite tree $T$ is
$\prod_{v\in T}\theta_{c(v)}$, and it is at least $p^{\abs{T}}$ whenever every offspring
weight along $T$ is at least $p$.
\end{lemma}

\begin{proof}
\leanok
Choose a field $e$ with sample $T$. We first show, by induction on the length of a word,
that any field $c$ satisfying $c(v)=e(v)$ for every $v\in T$ has the same sample as $e$.
Indeed, membership of a child $vj$ is determined by membership of $v$ and the single
inequality $j<c(v)$, so the induction never uses a coordinate outside $T$.

Since $T$ is finite, agreement with $e$ on $T$ is a finite cylinder event. Independence
of the coordinate field gives it mass
$\prod_{v\in T}\theta_{e(v)}$. This cylinder is contained in the event that the sample is
$T$. If every factor is at least $p$, its mass is at least $p^{\abs{T}}$, which proves the
point-mass bound.
\end{proof}

\begin{definition}[Truncated progeny]
\label{def:progeny}
\uses{def:sample}
\lean{BranchingProcess.progeny, BranchingProcess.progenyTree}
\leanok
The \emph{truncated progeny} $Z_k$ is the number of vertices of the sample within the first
$k$ generations.
\end{definition}

\begin{lemma}[The progeny recursion]
\label{lem:progeny-recursion}
\uses{def:progeny, lem:branching}
\lean{BranchingProcess.lintegral_pow_progeny_succ, BranchingProcess.lintegral_pow_progeny_le}
\leanok
Writing $F_k(s)=\bbE\bigl[s^{Z_k}\bigr]$, one has $F_{k+1}(s)=s\,f(F_k(s))$, and any
$y>1$ with $s f(y)=y$ bounds every $F_k(s)$.
\end{lemma}

\begin{proof}
\leanok
The truncations satisfy the deterministic recursion
\[
  Z_{k+1}(c)=1+\sum_{i<c(\rt)}Z_k(c_i),
\]
where $c_i(w)=c(iw)$ is the field below the $i$-th child. On the event
$c(\rt)=j$, the $j$ fields $c_i$ are independent copies of the original field. Therefore
\[
  \bbE\bigl[s^{Z_{k+1}}\,\mathbf1_{\set{c(\rt)=j}}\bigr]
  =s\theta_jF_k(s)^j.
\]
Summing over $j$ gives $F_{k+1}(s)=s f(F_k(s))$.

Since $Z_0=0$, we have $F_0(s)=1\leq y$. If $F_k(s)\leq y$, monotonicity of $f$ on the
nonnegative half-line gives
\[
  F_{k+1}(s)=s f(F_k(s))\leq s f(y)=y.
\]
Induction proves the uniform bound.
\end{proof}

\begin{lemma}[The progeny equation has a solution]
\label{lem:progeny-fixed}
\uses{lem:gen, def:offspring}
\lean{BranchingProcess.Offspring.exists_gen_lt_self,
BranchingProcess.Offspring.exists_progeny_fixedPoint}
\leanok
For a subcritical law there are $s,y>1$ with $s f(y)=y$.
\end{lemma}

\begin{proof}
\leanok
Use the factorisation from the proof of \cref{thm:supercritical},
\[
  1-f(t)=(1-t)g(t),
  \qquad g(1)=m<1.
\]
Continuity of $g$ gives $y>1$ sufficiently close to $1$ that $g(y)<1$. Since $1-y<0$,
the factorisation implies $f(y)<y$. On the other hand, monotonicity and $y>1$ give
$f(y)\geq f(1)=1$. Thus $s=y/f(y)$ is well defined and satisfies $s>1$ and
$s f(y)=y$.
\end{proof}

\begin{theorem}[Exponential moment of the total progeny]
\label{thm:progeny-moment}
\uses{lem:progeny-recursion, lem:progeny-fixed}
\lean{BranchingProcess.exists_exponential_moment,
BranchingProcess.exists_exponential_moment_bushMeasure}
\leanok
For a subcritical offspring law there is $\sigma>1$ with
$\bbE\bigl[\sigma^{\abs{\cT}}\bigr]<\infty$. The same holds for the size of a bush once the
conjugate law is subcritical.
\end{theorem}

\begin{proof}
\leanok
By \cref{lem:progeny-fixed}, choose $\sigma,y>1$ such that $\sigma f(y)=y$.
Then \cref{lem:progeny-recursion} gives
\[
  \bbE\bigl[\sigma^{Z_k}\bigr]\leq y
  \qquad\text{for every }k.
\]
The variables $Z_k$ increase with $k$. If the sample is finite, then $Z_k=\abs{\cT}$ for
all sufficiently large $k$, and hence
$\sigma^{\abs{\cT}}\leq\sup_k\sigma^{Z_k}$. If the sample is infinite, the same inequality
holds with Mathlib's convention that the finite cardinality of an infinite set is zero.
The monotone convergence theorem therefore gives
\[
  \bbE\bigl[\sigma^{\abs{\cT}}\bigr]
  \leq \sup_k\bbE\bigl[\sigma^{Z_k}\bigr]
  \leq y<\infty.
\]
For the bush law, apply the same result to the subcritical conjugate law and use
\cref{thm:harris-bushes} to identify the resulting Galton--Watson tree law with the law of
a bush.
\end{proof}
