\chapter{The Harris decomposition}
\label{ch:harris}

Conditioned on survival, the skeleton of a Galton--Watson tree is again a Galton--Watson
tree, for a reduced law, and the subtrees hanging off it are independent Galton--Watson trees
for a conjugate law. This chapter proves both halves and then their joint form.

\section{Conditioning at the root}

\begin{definition}[The conditioned laws]
\label{def:conditioned}
\uses{def:sampleMeasure}
\lean{BranchingProcess.survivalMeasure, BranchingProcess.bushMeasure,
BranchingProcess.skeletonDegree}
\leanok
Write $\bbP^{\mathrm{s}}$ for the field measure conditioned on survival and
$\bbP^{\mathrm{b}}$ for it conditioned on extinction. The \emph{skeleton degree} of a field
is the number of children of the root founding an infinite subtree.
\end{definition}

\begin{lemma}[The surviving-children count]
\label{lem:surviveWeight}
\uses{def:conditioned}
\lean{BranchingProcess.Offspring.surviveWeight, BranchingProcess.sampleMeasure_skeletonDegree,
BranchingProcess.Offspring.surviveWeight_zero}
\leanok
Unconditionally, the root has exactly $k$ surviving children with probability
\[
  \sum_{j}\theta_j\binom{j}{k}(1-q)^{k}q^{\,j-k},
\]
and the value at $k=0$ is $f(q)=q$.
\end{lemma}

\begin{proof}
\leanok
Condition on the root having $j$ children. For a fixed set $S$ of these children, the
branching property identifies the event that precisely the children in $S$ survive with a
product event for the $j$ shifted fields. Its probability is
\[
  \theta_j(1-q)^{\abs{S}}q^{\,j-\abs{S}}.
\]
The events obtained from distinct sets $S$ are disjoint. Summing over the
$\binom{j}{k}$ sets of size $k$, and then over $j$, gives the stated formula. When $k=0$,
the sum is $\sum_j\theta_jq^j=f(q)=q$ by
\cref{lem:extinction-fixed}.
\end{proof}

\begin{definition}[The reduced and the conjugate law]
\label{def:reduced}
\uses{lem:surviveWeight, lem:extinction-zero}
\lean{BranchingProcess.Offspring.reduced, BranchingProcess.Offspring.conjugate}
\leanok
The \emph{reduced law} is the Harris--Sevastyanov transform
$\tilde\theta_k=\bigl(\sum_j\theta_j\binom{j}{k}(1-q)^kq^{\,j-k}\bigr)/(1-q)$ for $k\geq1$ and
$\tilde\theta_0=0$; the \emph{conjugate law} is $\hat\theta_j=\theta_jq^{\,j}/q$. Both are
offspring laws with the same support bound.
\end{definition}

\begin{proof}
\leanok
All the displayed weights are nonnegative and vanish above $J$. To compute the total
surviving-child weight, interchange the two finite sums and apply the binomial theorem:
\[
  \sum_{k=0}^{J}\sum_{j=k}^{J}
  \theta_j\binom{j}{k}(1-q)^kq^{j-k}
  =\sum_{j=0}^{J}\theta_j\sum_{k=0}^{j}
  \binom{j}{k}(1-q)^kq^{j-k}
  =\sum_{j=0}^{J}\theta_j((1-q)+q)^j=1.
\]
By \cref{lem:surviveWeight}, the term with $k=0$ is $q$. The terms with $k\geq1$
therefore sum to $1-q$, and division by $1-q$ proves that the reduced weights sum to one.
For the conjugate weights,
\[
  \sum_{j=0}^{J}\hat\theta_j=q^{-1}\sum_{j=0}^{J}\theta_jq^j
  =q^{-1}f(q)=1.
\]
\end{proof}

\section{The two halves}

\begin{theorem}[The skeleton is a Galton--Watson tree]
\label{thm:harris-skeleton}
\uses{def:reduced, def:sampleMeasure}
\lean{BranchingProcess.skeletonTreeLaw_eq_treeLaw}
\leanok
If $J\leq N$ and $q<1$, the re-addressed skeleton of a sample conditioned on survival has the
law of the Galton--Watson tree of the reduced law.
\end{theorem}

\begin{proof}
\leanok
Re-address the skeleton by giving the $m$-th surviving child of each skeleton vertex the
letter $m$. Under the law conditioned on survival, \cref{lem:surviveWeight} shows that the
number of children of the root has law $\tilde\theta$. Conditional on this number, the
corresponding shifted fields are independent and each has the original law conditioned on
survival.

We apply this root identity recursively to a finite prefix-closed set $F$ of skeleton
addresses. Induction on the maximal length of a word in $F$ gives the same probability for
the containment event $\set{S\colon F\subseteq S}$ under the re-addressed skeleton law and
the Galton--Watson law with offspring distribution $\tilde\theta$. Replacing a finite set
of words by its prefix closure does not change this event for subtrees. These
containment events form a $\pi$-system that generates the measurable space of subtrees, so
uniqueness of probability measures proves the equality of laws.
\end{proof}

\begin{theorem}[The bushes are Galton--Watson trees]
\label{thm:harris-bushes}
\uses{def:reduced, def:sampleMeasure}
\lean{BranchingProcess.bushTreeLaw_eq_treeLaw}
\leanok
If $J\leq N$ and $q>0$, a sample conditioned on extinction has the law of the Galton--Watson
tree of the conjugate law.
\end{theorem}

\begin{proof}
\leanok
Conditioned on extinction, the root can have $j$ children only when all $j$ shifted
subfields become extinct. The branching property and independence therefore give
\[
  \bbP^{\mathrm b}(c(\rt)=j)
  =\frac{\theta_jq^j}{q}=\hat\theta_j.
\]
Conditional on this root count, the $j$ shifted fields are independent and each has law
$\bbP^{\mathrm b}$. Recursing over a finite prefix-closed set of addresses shows that the
bush law and the Galton--Watson law with offspring distribution $\hat\theta$ agree on every
finite containment event. As in the proof of \cref{thm:harris-skeleton}, these events
generate the measurable space of subtrees, which proves the equality of laws.
\end{proof}

\section{The joint form}

\begin{definition}[The decoration of a skeleton vertex]
\label{def:decoration}
\uses{thm:harris-skeleton, thm:harris-bushes}
\lean{BranchingProcess.decorationEvent, BranchingProcess.decorationMass}
\leanok
The \emph{decoration} of a skeleton vertex is its offspring count $j$, its number $k$ of
surviving children, and its dying subtrees indexed by rank.
\end{definition}

\begin{theorem}[The joint Harris decomposition]
\label{thm:harris-joint}
\uses{def:decoration}
\lean{BranchingProcess.survivalMeasure_decorations,
BranchingProcess.survivalMeasure_decorations_treeLaw}
\leanok
Conditioned on survival, probe the skeleton on a finite prefix-closed set $F$ of addresses and
constrain the decoration of each probed vertex. The mass factorises into one weight per
vertex,
\[
  \prod_{u\in F}
  \frac{\theta_{j_u}\binom{j_u}{k_u}(1-q)^{k_u}q^{\,j_u-k_u}}{1-q}
  \prod_{m<j_u-k_u}\bbP_{\hat\theta}(\cT\in\mathcal{A}_{u,m}),
\]
each dying subtree entering through the Galton--Watson law of the conjugate law.
\end{theorem}

\begin{proof}
\leanok
We first prove the factorisation at the root. Fix the root offspring count $j$ and the set
$S$ of its $k$ surviving children. Splitting the field at the root gives independent blocks:
the root count has mass $\theta_j$, the fields indexed by $S$ have law $\bbP^{\mathrm s}$,
and the remaining fields have law $\bbP^{\mathrm b}$. Summing over the
$\binom{j}{k}$ possible sets $S$ and dividing by the survival probability $1-q$ gives the
root decoration weight in the statement.

We now induct on the maximal length of an address in $F$. Since $F$ is prefix-closed, it
decomposes into the root and, for each child address occurring in $F$, a prefix-closed set
of suffixes. The compatibility assumption ensures that these children are among the first
$k_{\rt}$ surviving children. Apply the root factorisation, then apply the induction
hypothesis independently inside each of the corresponding surviving subfields. The
resulting products combine to give one decoration factor for every $u\in F$. Finally,
\cref{thm:harris-bushes} identifies the law of each dying subtree with the Galton--Watson
law of $\hat\theta$, yielding the displayed formula.
\end{proof}
