\chapter{The ambient tree}
\label{ch:tree}

The genealogy of a branching process is carried inside a fixed ambient tree, whose vertices
are words. Nothing in this chapter is probabilistic.

\begin{definition}[The ambient $N$-ary tree]
\label{def:word}
\lean{BranchingProcess.Word, BranchingProcess.wedge, BranchingProcess.treeDist}
\leanok
Fix $N\in\bbN$. A vertex of the ambient tree $\cN(N)$ is a finite word over the alphabet
$\{0,\dots,N-1\}$; the root is the empty word $\rt$ and the children of $v$ are the words
$v j$ with $j<N$. The \emph{wedge} $v\wedge w$ of two vertices is their longest common
prefix, and the graph metric of $\cN(N)$ is
\[
  d(v,w)=\abs{v}+\abs{w}-2\abs{v\wedge w}.
\]
\end{definition}

\begin{lemma}[The wedge is the greatest common prefix]
\label{lem:wedge}
\uses{def:word}
\lean{BranchingProcess.wedge_prefix_left, BranchingProcess.wedge_prefix_right,
BranchingProcess.prefix_wedge_of_prefix}
\leanok
The wedge $v\wedge w$ is a prefix of both $v$ and $w$, and every common prefix of $v$ and $w$
is a prefix of $v\wedge w$.
\end{lemma}

\begin{proof}
\leanok
We first prove that $v\wedge w$ is a prefix of $v$ by induction on $v$, splitting according
to whether $w$ is empty and, when both words are nonempty, whether their first letters
agree. The corresponding assertion for $w$ follows by symmetry. For maximality, we induct
on a common prefix $u$. If $u$ is nonempty, the prefix assumptions force $v$ and $w$ to have
the same first letter as $u$, and the claim reduces to their tails.
\end{proof}

\begin{lemma}[The metric]
\label{lem:treeDist}
\uses{def:word, lem:wedge}
\lean{BranchingProcess.treeDist_add, BranchingProcess.treeDist_eq_zero_iff,
BranchingProcess.treeDist_triangle}
\leanok
The subtraction defining $d$ does not truncate, $d(v,w)=0$ exactly when $v=w$, and $d$
satisfies the triangle inequality. A vertex and one of its children are at distance one.
\end{lemma}

\begin{proof}
\leanok
Since $v\wedge w$ is a prefix of both words, its length is bounded by each of
$\abs{v}$ and $\abs{w}$. Thus
$2\abs{v\wedge w}\leq\abs{v}+\abs{w}$, so the subtraction in the definition of $d$ does
not truncate. If $d(v,w)=0$, these two length bounds are equalities. A prefix of the same
length as a word is the word itself, which gives $v=v\wedge w=w$.

For the triangle inequality, compare the two prefixes $u\wedge v$ and $v\wedge w$ of $v$.
One is a prefix of the other, and the shorter one is consequently a common prefix of $u$
and $w$. Hence
\[
  \abs{u\wedge v}+\abs{v\wedge w}
  \leq \abs{v}+\abs{u\wedge w}.
\]
Substituting this bound into the three distance formulae gives
$d(u,w)\leq d(u,v)+d(v,w)$. Finally, if $w$ is a child of $v$, then $v$ is a prefix of $w$
and $\abs{w}=\abs{v}+1$, so $d(v,w)=1$.
\end{proof}

\begin{definition}[Subtrees]
\label{def:subtree}
\uses{def:word}
\lean{BranchingProcess.Subtree}
\leanok
A \emph{subtree} of $\cN(N)$ is a prefix-closed set of words, taken as a term of Mathlib's
\texttt{Descriptive.tree}. The residual subtree of $T$ at $v$ is
$T_v=\set{w : vw\in T}$.
\end{definition}
